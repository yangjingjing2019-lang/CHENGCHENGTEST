// 初始化時先載入一次語音清單（iOS 需要這個步驟） let systemVoices = []; function loadVoices() { if ('speechSynthesis' in window) { systemVoices = window.speechSynthesis.getVoices(); } } loadVoices(); if ('speechSynthesis' in window && window.speechSynthesis.onvoiceschanged !== undefined) { window.speechSynthesis.onvoiceschanged = loadVoices; } function playCurrentAudio() { if (!currentCorrectWord) return; if ('speechSynthesis' in window) { window.speechSynthesis.cancel(); let utterance = new SpeechSynthesisUtterance(currentCorrectWord); // 強制在系統語音中尋找普通話（Mandarin）或台灣國語的語音包 let targetVoice = systemVoices.find(voice => voice.lang === 'zh-CN' || voice.lang === 'zh-TW' || voice.lang === 'cmn-CN' ); if (targetVoice) { utterance.voice = targetVoice; } utterance.lang = 'zh-CN'; // 雙重保險 utterance.rate = 0.9; // 稍慢便於聽辨 window.speechSynthesis.speak(utterance); } }
